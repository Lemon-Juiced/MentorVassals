# ROOT scope pays the prestige cost based on highest held title tier.
mv_pay_reeducation_prestige_cost = {
	if = {
		limit = { highest_held_title_tier = tier_hegemony }
		add_prestige = -1000
	}
	else_if = {
		limit = { highest_held_title_tier = tier_empire }
		add_prestige = -800
	}
	else_if = {
		limit = { highest_held_title_tier = tier_kingdom }
		add_prestige = -600
	}
	else_if = {
		limit = { highest_held_title_tier = tier_duchy }
		add_prestige = -400
	}
	else = {
		add_prestige = -200
	}
}

# TARGET scope: repeatedly apply one-step upgrades until target reaches actor's tier.
mv_upgrade_education_to_actor_level = {
	while = {
		limit = {
			mv_can_be_raised_by_actor = yes
		}
		mv_upgrade_education_one_step = yes
	}
}

# TARGET scope: grant track-appropriate lifestyle XP.
mv_add_track_lifestyle_xp = {
	if = {
		limit = { mv_has_martial_education_track = yes }
		add_martial_lifestyle_xp = 250
	}
	else_if = {
		limit = { mv_has_martial_prowess_education_track = yes }
		add_martial_lifestyle_xp = 250
	}
	else_if = {
		limit = { mv_has_stewardship_education_track = yes }
		add_stewardship_lifestyle_xp = 250
	}
	else_if = {
		limit = { mv_has_diplomacy_education_track = yes }
		add_diplomacy_lifestyle_xp = 250
	}
	else_if = {
		limit = { mv_has_intrigue_education_track = yes }
		add_intrigue_lifestyle_xp = 250
	}
	else_if = {
		limit = { mv_has_learning_education_track = yes }
		add_learning_lifestyle_xp = 250
	}
	else_if = {
		limit = { mv_has_republican_knowledge_education_track = yes }
		add_stewardship_lifestyle_xp = 250
	}
}
