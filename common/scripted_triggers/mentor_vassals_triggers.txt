# This trigger is evaluated in recipient scope.
mv_same_education_track_as_actor = {
	OR = {
		AND = {
			scope:actor = { mv_has_martial_education_track = yes }
			mv_has_martial_education_track = yes
		}
		AND = {
			scope:actor = { mv_has_martial_prowess_education_track = yes }
			mv_has_martial_prowess_education_track = yes
		}
		AND = {
			scope:actor = { mv_has_stewardship_education_track = yes }
			mv_has_stewardship_education_track = yes
		}
		AND = {
			scope:actor = { mv_has_diplomacy_education_track = yes }
			mv_has_diplomacy_education_track = yes
		}
		AND = {
			scope:actor = { mv_has_intrigue_education_track = yes }
			mv_has_intrigue_education_track = yes
		}
		AND = {
			scope:actor = { mv_has_learning_education_track = yes }

	}

# Full recipient gate; evaluated in recipient scope.
mv_is_valid_reeducation_target = {
	is_adult = yes
	is_alive = yes
	OR = {
		is_vassal_or_below_of = scope:actor
		is_child_of = scope:actor
	}
	NOT = { has_character_flag = mv_recently_reeducated }
	mv_has_any_education_track = yes
	scope:actor = { mv_has_any_education_track = yes }
	mv_same_education_track_as_actor = yes
	mv_can_be_raised_by_actor = yes
}
